calculations:
import numpy as np
from engineering_notation import EngNumber
# Given values
H = 70 # Heart rate in beats per minute
Vs = 6 # Supply voltage in volts
Switching_factor = 0.68
Vo = Switching_factor * Vs
C = 47e-6 # Capacitance in farads
# Calculate the required resistor value
R = -60 / (H * C * np.log(1 - (Vo / Vs)))
print('R={0}'.format(EngNumber(R)))
